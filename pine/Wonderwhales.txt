f_sessionOk(string sess, string tz) =>
    not na(time(timeframe.period, sess, tz))

f_vsaColor(float vol, float ma, float rP, float rVH, float rH, float rN, float rL) =>
    vol >= ma * rP ? color.purple : vol >= ma * rVH ? color.red : vol >= ma * rH ? color.orange : vol >= ma * rN ? color.green : vol >= ma * rL ? color.blue : color.silver

f_vsaName(float vol, float ma, float rP, float rVH, float rH, float rN, float rL) =>
    vol >= ma * rP ? "TIM" : vol >= ma * rVH ? "VeryHigh" : vol >= ma * rH ? "High" : vol >= ma * rN ? "Normal" : vol >= ma * rL ? "Low" : "VeryLow"

// --- Pure math / formatting (truoc do copy 3 ban trong parts/, parts-xau/, parts-xau2/) ---
f_clamp(float v, float lo, float hi) =>
    math.max(lo, math.min(hi, v))

f_fmtPx(float p) =>
    str.tostring(p, format.mintick)

f_isNewDay() =>
    ta.change(time("D")) != 0

// CANH BAO: str.format_time(time, format, timezone) - `time` (series int) di TRUOC.
// Dau tien la UNIX timestamp, thu hai moi la chuoi format. Dao nguoc -> CE10123
// "argument of literal string type was used but a series int is expected".
f_ts() =>
    str.format_time(time, "yyyy-MM-dd'T'HH:mm:ss'Z'", "UTC")

f_sym() =>
    str.replace_all(syminfo.ticker, "\\", "/")

f_sessAsia(string tz) =>
    not na(time(timeframe.period, "1700-0200", tz))

f_sessLondon(string tz) =>
    not na(time(timeframe.period, "0200-1000", tz))

f_sessNy(string tz) =>
    not na(time(timeframe.period, "0700-1500", tz))

f_sessName(bool asia, bool london, bool ny) =>
    london and ny ? "London/NY" : ny ? "New York" : london ? "London" : asia ? "Asia" : "-"

slowEmaSafe = math.max(fastEmaSafe + 1, ww_slowEmaLen)
ema200Safe  = math.max(10, ww_ema200Len)

// Ensure ATR period is strictly positive
atrLenSafe = math.max(1, ww_atrLen)

// Ensure Volume MA and Secondary indicators have positive lookbacks

rsiLenSafe   = math.max(1, ww_rsiLen)

// --- Parabolic SAR ---
ww_psar = ta.sar(ww_psarStart, ww_psarInc, psarMaxSafe)

// Direction tracking: 1 = Bullish (SAR below price), -1 = Bearish (SAR above price)
var int ww_sarDir = 0
if not na(ww_psar)
    ww_sarDir := ww_psar < close ? 1 : -1

ww_sarBull = ww_sarDir == 1
ww_sarBear = ww_sarDir == -1

// Flip triggers: fire strictly on the transition bar
ww_buyFlip  = ww_sarBull and ww_sarDir[1] == -1
ww_sellFlip = ww_sarBear and ww_sarDir[1] == 1

// --- Volume Spread Analysis (VSA) Engine ---
ww_vol          = nz(volume)
ww_volMa        = ta.sma(ww_vol, volMaLenSafe)
ww_volCol       = f_vsaColor(ww_vol, ww_volMa, ww_volSpikeR, ww_volVeryHighR, ww_volHighR, ww_volNormalR, ww_volLowR)
ww_volBucket    = f_vsaName(ww_vol, ww_volMa, ww_volSpikeR, ww_volVeryHighR, ww_volHighR, ww_volNormalR, ww_volLowR)
ww_volRatio     = not na(ww_volMa) and ww_volMa > 0.0 ? ww_vol / ww_volMa : 0.0
ww_isClimaxVol  = not na(ww_volMa) and ww_volMa > 0.0 and ww_vol >= ww_volMa * ww_volSpikeR
ww_isHighVol    = not na(ww_volMa) and ww_volMa > 0.0 and ww_vol >= ww_volMa * ww_volHighR

// --- Exponential Moving Averages ---
ww_fastEma      = ta.ema(close, fastEmaSafe)
ww_slowEma      = ta.ema(close, slowEmaSafe)
ww_emaBull      = ww_fastEma > ww_slowEma
ww_emaBear      = ww_fastEma < ww_slowEma
ww_emaBuyCross  = ta.crossover(ww_fastEma, ww_slowEma)
ww_emaSellCross = ta.crossunder(ww_fastEma, ww_slowEma)

// 200 EMA Baseline
ww_ema200       = ta.ema(close, ema200Safe)
ww_aboveEma200  = close > ww_ema200

// --- Average True Range & Percentage Volatility ---
ww_atr          = ta.atr(atrLenSafe)
ww_atrPct       = close > 0 and not na(ww_atr) ? (ww_atr / close) * 100.0 : 0.0
ww_volatilityOk = ww_atrPct >= ww_minAtrPct

// --- Relative Strength Index (RSI) ---
ww_rsi           = ta.rsi(close, rsiLenSafe)
ww_rsiOverbought = ww_rsi >= 70
ww_rsiOversold   = ww_rsi <= 30
ww_rsiBullish    = ww_rsi > 50

// --- Bollinger Bands ---
[ww_bbMid, ww_bbUpper, ww_bbLower] = ta.bb(close, bbLenSafe, ww_bbMult)

// =============================================================================
//  WONDERWHALES — SIGNAL LOGIC & FILTER GATES
//  Pure signal structure: 100% preserved from original v1.0 specification

bool ww_rawBuy      = false
bool ww_rawSell     = false

bool ww_validSell   = false
bool ww_blockedBuy  = false

    ww_rawBuy    := ww_buyFlip
    ww_rawSell   := ww_sellFlip
    ww_validBuy  := ww_rawBuy
    ww_validSell := ww_rawSell

else if ww_mode == "EMA Crossover"
    ww_rawBuy    := ww_emaBuyCross
    ww_rawSell   := ww_emaSellCross
    ww_validBuy  := ww_rawBuy
    ww_validSell := ww_rawSell

else if ww_mode == "SAR + EMA Trend Filter"
    ww_rawBuy      := ww_buyFlip
    ww_rawSell     := ww_sellFlip
    ww_validBuy    := ww_rawBuy and ww_emaBull
    ww_validSell   := ww_rawSell and ww_emaBear
    ww_blockedBuy  := ww_rawBuy and not ww_emaBull
    ww_blockedSell := ww_rawSell and not ww_emaBear

else if ww_mode == "SAR + EMA + ATR Filter"
    ww_rawBuy      := ww_buyFlip
    ww_rawSell     := ww_sellFlip
    bool emaOkBuy  = ww_emaBull
    bool emaOkSell = ww_emaBear
    bool atrOk     = ww_volatilityOk
    ww_validBuy    := ww_rawBuy and emaOkBuy and atrOk
    ww_validSell   := ww_rawSell and emaOkSell and atrOk
    ww_blockedBuy  := ww_rawBuy and not (emaOkBuy and atrOk)
    ww_blockedSell := ww_rawSell and not (emaOkSell and atrOk)

// Anti-repainting bar confirmation gate

ww_signalBuy         = ww_validBuy and ww_barConfirmed
ww_signalSell        = ww_validSell and ww_barConfirmed
ww_signalBlockedBuy  = ww_blockedBuy and ww_barConfirmed
ww_signalBlockedSell = ww_blockedSell and ww_barConfirmed
