// =============================================================================
//  S/R ZONE ENGINE - dung chung parts-xau + parts-xau2
//  Pivot-based supply/demand zones, ATR width, zone merging, max limit.
//  Truoc do copy 2 ban (SrZone/xau_* vs X2SrZone/x2_*) - gop thanh 1.
//  Bien mang dung chung `sr_zones` (moi dist la 1 script doc lap nen khong xung dot);
//  ham nhan mang + ATR/width/maxZone lam tham so (khong tham khao bien toan cuc
//  vi shared duoc chen TRUOC 10_config/20_core).
// =============================================================================

type SrZone
    float top
    float bot
    int   side      // 1=support, -1=resistance
    int   touches
    bool  broken

var array<SrZone> sr_zones = array.new<SrZone>()

f_addZone(array<SrZone> zones, float px, int side, float atr, float width, int maxZone) =>
    float half   = atr * width * 0.5
    bool  merged = false
    for z in zones
        if math.abs((z.top + z.bot) / 2 - px) < atr * width
            z.touches += 1
            merged     := true
            break
    if not merged
        array.push(zones, SrZone.new(px + half, px - half, side, 1, false))
    for i = array.size(zones) - 1 to 0
        SrZone z2 = array.get(zones, i)
        if z2.broken
            array.remove(zones, i)
    while array.size(zones) > maxZone
        array.shift(zones)

f_nearSupport(array<SrZone> zones, float atr) =>
    bool found = false
    for z in zones
        if not z.broken and z.side == 1 and low >= z.bot - atr * 0.3 and low <= z.top + atr * 0.3
            found := true
    found

f_nearResistance(array<SrZone> zones, float atr) =>
    bool found = false
    for z in zones
        if not z.broken and z.side == -1 and high >= z.bot - atr * 0.3 and high <= z.top + atr * 0.3
            found := true
    found

// =============================================================================
//  SHARED SIGNAL QUALITY PRIMITIVES - used by pine/parts-tm2 (TM Indicator 2)
//  Every helper here is PURE: no reference to inputs or globals, because the
//  SHARED block is inserted before 01_inputs_*.pine. This is the single source of
//  truth for the v2.1 direction-aware scoring contract:
//
//    * direction and strength are separate values;
//    * LONG and SHORT scores use the same 0..100 scale and the same thresholds;
//    * a signed feature f in [-1, 1] contributes max(f, 0) to LONG and
//      max(-f, 0) to SHORT, so the two sides are mathematically symmetric;
//    * gate/risk failures are reason codes, never score compensation.
// =============================================================================

f_tmClamp(float v, float lo, float hi) =>
    math.max(lo, math.min(hi, v))

// Directional support of a signed feature: one side gets the magnitude, the
// other gets 0. No sign flipping, no negative quality score.
f_tmLongPart(float v) =>
    math.max(v, 0.0)

f_tmShortPart(float v) =>
    math.max(-v, 0.0)

// Aggregate one side to 0..100. `weighted` is the sum of w * directional part,
// `totalWeight` the sum of all directional weights actually enabled.
f_tmSideScore(float weighted, float totalWeight) =>
    totalWeight > 0.0 ? math.round(100.0 * f_tmClamp(weighted / totalWeight, 0.0, 1.0)) : 0

// 0 = NO TRADE, 1 = WEAK, 2 = SETUP, 3 = STRONG, 4 = A+
f_tmScoreLevel(int score, int noTrade, int weak, int setup, int strong) =>
    score <= noTrade ? 0 : score <= weak ? 1 : score <= setup ? 2 : score <= strong ? 3 : 4

// =============================================================================
//  L2 - VOLATILITY & MARKET REGIME CLASSIFIER (Phase 3 Core Engine)

// --- 1. Basic Volatility Calculations ----------------------------------------
tm_atr    = ta.atr(tm_atrLen)
tm_atrPct = not na(close) and close > 0.0 ? (tm_atr / close) * 100.0 : 0.0

// Bollinger Bands & Squeeze Measurement
tm_bbBasis = ta.sma(close, 20)
tm_bbDev   = 2.0 * ta.stdev(close, 20)
tm_bbUpper = tm_bbBasis + tm_bbDev
tm_bbLower = tm_bbBasis - tm_bbDev
tm_bbWidth = tm_bbBasis > 0.0 ? (tm_bbUpper - tm_bbLower) / tm_bbBasis * 100.0 : 0.0

tm_bbwLowest  = ta.lowest(tm_bbWidth, tm_regimeBbwLen)
tm_bbwHighest = ta.highest(tm_bbWidth, tm_regimeBbwLen)
tm_bbwPct     = (tm_bbwHighest - tm_bbwLowest) > 0.0 ? (tm_bbWidth - tm_bbwLowest) / (tm_bbwHighest - tm_bbwLowest) * 100.0 : 50.0
bool tm_isSqueeze = tm_bbwPct <= tm_regimeBbwSqueezePct
bool tm_bbExpand  = tm_bbWidth > ta.sma(tm_bbWidth, 20)

// Directional Movement & ADX
[tm_diPlus, tm_diMinus, tm_adx] = ta.dmi(tm_dmiLen, 14)

// EMA 20 and 50 Slopes (normalized by ATR to be asset-neutral)
tm_ema20 = ta.ema(close, tm_emaFast)
tm_ema50 = ta.ema(close, tm_emaMid)
float tm_emaSlope20 = not na(tm_atr) and tm_atr > 0.0 ? (tm_ema20 - tm_ema20[3]) / (3.0 * tm_atr) : 0.0
float tm_emaSlope50 = not na(tm_atr) and tm_atr > 0.0 ? (tm_ema50 - tm_ema50[5]) / (5.0 * tm_atr) : 0.0

bool tm_emaBull = close > tm_ema20 and tm_ema20 > tm_ema50 and tm_emaSlope20 > 0.03
bool tm_emaBear = close < tm_ema20 and tm_ema20 < tm_ema50 and tm_emaSlope20 < -0.03

// --- 2. Market Regime Classifier ---------------------------------------------
// Phân loại thị trường thành:
//  - TRENDING (1): ADX >= adxTrending, EMA aligned, BBW expanding, không ở trong Squeeze.
//  - CHOPPY_SQUEEZE (-1): ADX < adxChop hoặc BBW trong vùng nén chặt (Squeeze).
//  - RANGING (0): Thị trường tích lũy / đi ngang với biên độ rõ ràng, thích hợp đánh Mean-Reversion tại S/R.
f_classifyRegime(float adxVal, bool isSq, float slope20, bool emaUp, bool emaDn, float bbwP) =>
    int regCode  = 0
    int tDir     = 0
    string rName = "RANGING"

    if adxVal >= tm_regimeAdxTrending and (emaUp or emaDn) and not isSq
        regCode := 1
        tDir    := emaUp ? 1 : -1
        rName   := emaUp ? "TRENDING_BULL" : "TRENDING_BEAR"
    else if adxVal < tm_regimeAdxChop or (isSq and math.abs(slope20) < 0.05)
        regCode := -1
        tDir    := 0
        rName   := "CHOPPY_SQUEEZE"
    else
        regCode := 0
        tDir    := 0
        rName   := "RANGING"

    MarketRegime.new(regCode, tDir, rName, adxVal, bbwP, isSq)

tm_regime = f_classifyRegime(tm_adx, tm_isSqueeze, tm_emaSlope20, tm_emaBull, tm_emaBear, tm_bbwPct)

// Volatility score [-1, +1]. v2.1 keeps volatility OUT of the directional score:
// it drives the regime classifier, mode selection and the gates instead.
float tm_volScore = tm_bbExpand ? 1.0 : tm_isSqueeze ? -1.0 : 0.0

// =============================================================================
//  L3 - MARKET STRUCTURE, PRICE ACTION & LIQUIDITY (v2.0)
// =============================================================================

// --- 1. Basic Candle Dimensions & Metrics ------------------------------------
float tm_body    = math.abs(close - open)
float tm_range   = math.max(high - low, syminfo.mintick)
float tm_hiWick  = high - math.max(close, open)
float tm_loWick  = math.min(close, open) - low
bool  tm_bullBar = close > open
bool  tm_bearBar = close < open

float tm_prevBody = math.abs(close[1] - open[1])

// Displacement Candle
bool tm_dispBull = tm_body >= 1.5 * tm_atr and tm_hiWick < tm_body * 0.3
bool tm_dispBear = tm_body >= 1.5 * tm_atr and tm_loWick < tm_body * 0.3

// Pin Bar Pattern
bool tm_bullPin = tm_bullBar and tm_loWick >= 2.0 * tm_body
bool tm_bearPin = tm_bearBar and tm_hiWick >= 2.0 * tm_body

// Engulfing Pattern
bool tm_bullEng = tm_bullBar and close[1] < open[1] and close >= open[1] and open <= close[1] and tm_body >= tm_prevBody
bool tm_bearEng = tm_bearBar and close[1] > open[1] and close <= open[1] and open >= close[1] and tm_body >= tm_prevBody

// Rejection Candle
bool tm_bullRej = tm_hiWick >= tm_range * 0.55 and tm_loWick <= tm_range * 0.15 and tm_bullBar
bool tm_bearRej = tm_loWick >= tm_range * 0.55 and tm_hiWick <= tm_range * 0.15 and tm_bearBar

float tm_cpBull = (tm_bullPin ? 1.0 : 0.0) + (tm_bullEng ? 1.0 : 0.0) + (tm_dispBull ? 1.2 : 0.0) + (tm_bullRej ? 0.8 : 0.0)
float tm_cpBear = (tm_bearPin ? 1.0 : 0.0) + (tm_bearEng ? 1.0 : 0.0) + (tm_dispBear ? 1.2 : 0.0) + (tm_bearRej ? 0.8 : 0.0)
float tm_cpScore = tm_cpBull > tm_cpBear ? math.min(1.0, tm_cpBull / 3.0) : tm_cpBear > tm_cpBull ? -math.min(1.0, tm_cpBear / 3.0) : 0.0

// --- 2. Event Buffer & Helper ------------------------------------------------
var array<TMEvent> tm_evtBuf = array.new<TMEvent>()

f_pushEvent(string kind, string dir, int barIdx, int timeVal, bool confirmed, float lvl) =>
    array.push(tm_evtBuf, TMEvent.new(kind, dir, timeVal, barIdx, timeframe.period, confirmed, lvl))
    while array.size(tm_evtBuf) > 50
        array.shift(tm_evtBuf)

// --- 3. Swings & Trend Tracking ----------------------------------------------
float tm_pivH = ta.pivothigh(high, tm_ms_swLen, tm_ms_swLen)
float tm_pivL = ta.pivotlow(low,   tm_ms_swLen, tm_ms_swLen)

var float tm_swingHi  = na
var float tm_swingLo  = na
var float tm_prevHigh = na
var float tm_prevLow  = na
var int   tm_hiBar    = na
var int   tm_loBar    = na

if not na(tm_pivH)
    tm_prevHigh := tm_swingHi
    tm_swingHi  := high[tm_ms_swLen]
    tm_hiBar    := bar_index - tm_ms_swLen

if not na(tm_pivL)
    tm_prevLow := tm_swingLo
    tm_swingLo := low[tm_ms_swLen]
    tm_loBar   := bar_index - tm_ms_swLen

var int tm_trend = 0
if not na(tm_swingHi) and not na(tm_prevHigh) and not na(tm_swingLo) and not na(tm_prevLow)
    if tm_swingHi > tm_prevHigh and tm_swingLo > tm_prevLow
        tm_trend := 1
    else if tm_swingHi < tm_prevHigh and tm_swingLo < tm_prevLow
        tm_trend := -1

// --- 4. BOS, CHoCH, MSS ------------------------------------------------------
bool tm_crossOverHi  = ta.crossover(close, tm_swingHi)
bool tm_crossUnderLo = ta.crossunder(close, tm_swingLo)

bool tm_bosUp   = tm_ms_bosOn and not na(tm_swingHi) and tm_crossOverHi and tm_trend == 1
bool tm_bosDown = tm_ms_bosOn and not na(tm_swingLo) and tm_crossUnderLo and tm_trend == -1

bool tm_chochUp   = tm_ms_chochOn and not na(tm_swingHi) and tm_crossOverHi and tm_trend == -1
bool tm_chochDown = tm_ms_chochOn and not na(tm_swingLo) and tm_crossUnderLo and tm_trend == 1

bool tm_mssUp   = tm_ms_mssOn and tm_chochUp and tm_dispBull
bool tm_mssDown = tm_ms_mssOn and tm_chochDown and tm_dispBear

if tm_bosUp
    f_pushEvent("BOS", "bull", bar_index, time, barstate.isconfirmed, tm_swingHi)
if tm_bosDown
    f_pushEvent("BOS", "bear", bar_index, time, barstate.isconfirmed, tm_swingLo)
if tm_chochUp
    f_pushEvent("CHOCH", "bull", bar_index, time, barstate.isconfirmed, tm_swingHi)
if tm_chochDown
    f_pushEvent("CHOCH", "bear", bar_index, time, barstate.isconfirmed, tm_swingLo)
if tm_mssUp
    f_pushEvent("MSS", "bull", bar_index, time, barstate.isconfirmed, tm_swingHi)
if tm_mssDown
    f_pushEvent("MSS", "bear", bar_index, time, barstate.isconfirmed, tm_swingLo)

// --- 5. Retest & Failed Breakout ---------------------------------------------
var float tm_lastBreakLvl = na
var int   tm_lastBreakBar = na
var int   tm_lastBreakDir = 0

if tm_bosUp or tm_chochUp
    tm_lastBreakLvl := tm_swingHi
    tm_lastBreakBar := bar_index
    tm_lastBreakDir := 1
else if tm_bosDown or tm_chochDown
    tm_lastBreakLvl := tm_swingLo
    tm_lastBreakBar := bar_index
    tm_lastBreakDir := -1

// Liquidity Sweep Detection (v1 Parity)

bool tm_sweepBear = (tm_liq_pdOn and high > tm_pdh and close < tm_pdh) or (tm_liq_pwOn and high > tm_pwh and close < tm_pwh) or (tm_liq_sessOn and not na(tm_sessHi) and high > tm_sessHi and close < tm_sessHi) or (tm_eqHi and high[tm_ms_swLen] > tm_prevHigh and close < tm_prevHigh)

// window. Storing the swept extreme lets Mode B invalidate itself instead of
// triangulating a level from PDH/PDL that may have changed since the sweep.
var int   tm_lastSweepBullBar = -1
var int   tm_lastSweepBearBar = -1

var float tm_lastSweepBearPx  = na

if tm_sweepBull
    tm_lastSweepBullBar := bar_index
    tm_lastSweepBullPx  := low
    f_pushEvent("SWEEP", "bull", bar_index, time, barstate.isconfirmed, math.max(tm_pdl, nz(tm_sessLo, tm_pdl)))
if tm_sweepBear
    tm_lastSweepBearBar := bar_index
    tm_lastSweepBearPx  := high
    f_pushEvent("SWEEP", "bear", bar_index, time, barstate.isconfirmed, math.min(tm_pdh, nz(tm_sessHi, tm_pdh)))

// --- 3. BTC Leader Filter for ETH / Altcoin (Request 5) ----------------------

bool tm_needBtcLeader = tm_btcLeaderOn and not tm_isBtcSelf and (tm_assetMode == "ETH" or tm_assetMode == "Liquid Altcoin" or tm_assetMode == "High-Risk Altcoin" or tm_assetMode == "Auto")

[btcClose, btcEmaF, btcRsi, btcAtr] = request.security(tm_btcTicker, timeframe.period, [close[1], ta.ema(close, 20)[1], ta.rsi(close, 14)[1], ta.atr(14)[1]], lookahead = barmerge.lookahead_on, gaps = barmerge.gaps_off)

bool btcPumping = tm_needBtcLeader and not na(btcClose) and not na(btcEmaF) and not na(btcAtr) and (btcClose > btcEmaF + tm_btcImpulseAtr * btcAtr or btcRsi > 68.0)

// BTC Gate: Altcoin LONG bị chặn nếu BTC đang xả mạnh; Altcoin SHORT bị chặn nếu BTC đang bơm mạnh
bool tm_btcLongOk  = not tm_needBtcLeader or not btcDumping
bool tm_btcShortOk = not tm_needBtcLeader or not btcPumping

// --- 4. Weekend Low-Liquidity Filter (Crypto) --------------------------------
bool tm_isWeekend = dayofweek == dayofweek.saturday or dayofweek == dayofweek.sunday
bool tm_isCrypto = syminfo.type == "crypto" or tm_assetMode == "BTC" or tm_assetMode == "ETH" or tm_assetMode == "Liquid Altcoin" or tm_assetMode == "High-Risk Altcoin" or str.contains(syminfo.ticker, "USDT")
bool tm_weekendBlocked = tm_weekendFilter and tm_isCrypto and tm_isWeekend and (tm_volRatio < 0.8 or tm_atrPct < 0.6)

// =============================================================================
//  L6 - ROOM-TO-LEFT / OBSTACLE GATE & ELIGIBILITY EVALUATOR (v2.0)
// =============================================================================

// --- 1. Nearest Barrier / Obstacle Calculation (Room-to-Left) ----------------
float tm_nearestRes = na
float tm_nearestSup = na

// Scan unbroken S/R zones
for z in sr_zones
    if not z.broken
        if z.side == -1 and z.bot > close
            tm_nearestRes := na(tm_nearestRes) ? z.bot : math.min(tm_nearestRes, z.bot)
        if z.side == 1 and z.top < close
            tm_nearestSup := na(tm_nearestSup) ? z.top : math.max(tm_nearestSup, z.top)

// v2.1 Bug 2: FVGs are intentionally NOT obstacles for this gate. A fresh FVG
// can sit within a fraction of an ATR of price and would recreate the same
// false blocks as the old 5-bar swing scan. FVGs are still used as LIMIT entry
// locations in the plan builder.

// Scan active Order Blocks
for o in tm_obs
    if not o.mitigated
        if o.side == -1 and o.bot > close
            tm_nearestRes := na(tm_nearestRes) ? o.bot : math.min(tm_nearestRes, o.bot)
        if o.side == 1 and o.top < close
            tm_nearestSup := na(tm_nearestSup) ? o.top : math.max(tm_nearestSup, o.top)

// Factor in PDH / PDL and PWH / PWL only when sufficiently distant from entry (> 0.3 * tm_atr)
float tm_minObsDist = nz(tm_atr, close * 0.01) * 0.3
if not na(tm_pdh) and tm_pdh > close and (tm_pdh - close) >= tm_minObsDist
    tm_nearestRes := na(tm_nearestRes) ? tm_pdh : math.min(tm_nearestRes, tm_pdh)
if not na(tm_pdl) and tm_pdl < close and (close - tm_pdl) >= tm_minObsDist
    tm_nearestSup := na(tm_nearestSup) ? tm_pdl : math.max(tm_nearestSup, tm_pdl)

if not na(tm_pwh) and tm_pwh > close and (tm_pwh - close) >= tm_minObsDist
    tm_nearestRes := na(tm_nearestRes) ? tm_pwh : math.min(tm_nearestRes, tm_pwh)
if not na(tm_pwl) and tm_pwl < close and (close - tm_pwl) >= tm_minObsDist
    tm_nearestSup := na(tm_nearestSup) ? tm_pwl : math.max(tm_nearestSup, tm_pwl)

// Distance to Obstacle vs Target Distance to TP1
float tm_estDistTp1 = nz(tm_atr, close * 0.01) * tm_sl_atrMul * tm_rr1

float tm_distToRes = not na(tm_nearestRes) ? tm_nearestRes - close : na
float tm_distToSup = not na(tm_nearestSup) ? close - tm_nearestSup : na

bool tm_obstacleBlockedLong  = tm_gate_obstacle and not na(tm_distToRes) and tm_distToRes < (tm_estDistTp1 * tm_obstacleBuffer)
bool tm_obstacleBlockedShort = tm_gate_obstacle and not na(tm_distToSup) and tm_distToSup < (tm_estDistTp1 * tm_obstacleBuffer)

// --- 2. Smart Eligibility Gates Evaluation -----------------------------------
// v2.1: every blocker produces a standardized reason code. `rsn` keeps the
// highest-priority blocker (stable order, used by the compact dashboard), while
// `mask` keeps ALL active blockers for the Advanced dashboard and alert payload.
// Score/quality never appears here: gates are independent of confidence.
f_evaluateGates(int dir) =>
    bool   regimeOk   = true
    bool   mtfOk      = true
    bool   btcOk      = true
    bool   obstacleOk = true
    bool   overextOk  = true
    bool   circuitOk  = true
    string rsn        = "OK"
    string mask       = ""

    // 1. Regime Gate: no breakout entries in CHOPPY/SQUEEZE, and no buying the
    // upper half / selling the lower half of a RANGING box.
    bool regimeBlocked = false
    if tm_gate_regime and tm_regime.code == -1
        regimeBlocked := true
    else if tm_gate_regime and tm_regime.code == 0
        if not na(tm_nearestRes) and not na(tm_nearestSup)
            float rangeMid = (tm_nearestRes + tm_nearestSup) * 0.5
            if dir == 1 and close > rangeMid
                regimeBlocked := true
            if dir == -1 and close < rangeMid
                regimeBlocked := true
    if regimeBlocked
        regimeOk := false
        rsn      := "BLOCKED_REGIME"
        mask     := f_tmBlockers(mask, "BLOCKED_REGIME")

    // 2. MTF hierarchy + Context veto + BTC leader
    bool setupOk = not tm_htfFilter or not tm_tfHierarchyOk or tm_setupDir == 0 or tm_setupDir == dir
    bool vetoCtx = tm_htfVetoOn and tm_tfHierarchyOk and tm_ctxDir != 0 and tm_ctxDir != dir
    mtfOk := setupOk and not vetoCtx
    if not mtfOk
        if rsn == "OK"
            rsn := "BLOCKED_MTF"
        mask := f_tmBlockers(mask, "BLOCKED_MTF")

    btcOk := dir == 1 ? tm_btcLongOk : dir == -1 ? tm_btcShortOk : true
    if not btcOk
        string btcCode = dir == 1 ? "BTC_DUMPING_VETO" : "BTC_PUMPING_VETO"
        if rsn == "OK"
            rsn := btcCode
        mask := f_tmBlockers(mask, btcCode)

    // 3. Obstacle Gate (Room-to-Left)
    obstacleOk := dir == 1 ? not tm_obstacleBlockedLong : dir == -1 ? not tm_obstacleBlockedShort : true
    if not obstacleOk
        if rsn == "OK"
            rsn := "BLOCKED_OBSTACLE"
        mask := f_tmBlockers(mask, "BLOCKED_OBSTACLE")

    // 4. Overextension Gate (market entries only; Auto mode converts to limit)
    overextOk := not tm_gate_overext or not tm_isOverextended or tm_entryMode != "Market on Confirm"
    if not overextOk
        if rsn == "OK"
            rsn := "BLOCKED_OVEREXTENSION"
        mask := f_tmBlockers(mask, "BLOCKED_OVEREXTENSION")

    // 5. Weekend / Low Liquidity Filter
    if tm_weekendBlocked
        circuitOk := false
        if rsn == "OK"
            rsn := "BLOCKED_LIQUIDITY"
        mask := f_tmBlockers(mask, "BLOCKED_LIQUIDITY")

    bool allPassed = regimeOk and mtfOk and btcOk and obstacleOk and overextOk and circuitOk
    GateStatus.new(regimeOk, mtfOk, btcOk, obstacleOk, overextOk, circuitOk, allPassed, rsn, mask)

// =============================================================================
//  L7 - DIRECTION-AWARE SCORING ENGINE (v2.1)
// =============================================================================
//  Roadmap v2.1 P0: the v2.0 engine mapped one signed raw score to a 0-100
//  quality score (score100 = 50 + raw * 50) and then compared it to a single
//  no-trade threshold. Bearish evidence therefore produced a LOW score that
//  could never pass the quality gate, so SHORT setups were blocked by design.
//
//  Contract implemented here:
//    1. Every group emits a signed value f in [-1, 1] (direction + magnitude).
//    2. LONG consumes max(f, 0), SHORT consumes max(-f, 0): identical weights,
//       identical scale, identical thresholds.
//    3. Side score = 100 * weightedSupport / enabledDirectionalWeight, so a
//       neutral group dilutes BOTH sides and cannot favour one of them.
//    4. Direction needs a minimum side score and a margin; otherwise the state
//       is NO_DIRECTION / AMBIGUOUS_SETUP and no side is forced.
//    5. Regime/volatility is not a directional feature: it drives mode + gates.
//       No hidden "choppy damping" multiplier that would make scores
//       incomparable between regimes.
// =============================================================================

string tm_scoreVersion    = "2.1"
string tm_scoreFeatureSet = "structure+location+trend+momentum+volume+mtf"

// --- 1. Normalized Group Scores [-1.0 to +1.0] -------------------------------
float tm_gTrend  = tm_trendScore
float tm_gStruct = (tm_bosUp or tm_chochUp or tm_mssUp) ? 1.0 : (tm_bosDown or tm_chochDown or tm_mssDown) ? -1.0 : float(tm_trend) * 0.5
float tm_gLiq    = tm_sweepBull ? 1.0 : tm_sweepBear ? -1.0 : tm_retestBull ? 0.7 : tm_retestBear ? -0.7 : 0.0
float tm_gLoc    = tm_vwapScore
float tm_gSr     = f_nearSupport(sr_zones, tm_atr) ? 1.0 : f_nearResistance(sr_zones, tm_atr) ? -1.0 : not na(tm_nearFvgBullPx) ? 0.8 : not na(tm_nearFvgBearPx) ? -0.8 : 0.0
float tm_gCandle = tm_cpScore
float tm_gVolume = tm_volumeScore
float tm_gMom    = tm_momScore
float tm_gSess   = (close >= open ? 1.0 : -1.0) * (tm_cryptoSessionNeutral ? 0.0 : 0.5)
float tm_gMtf    = tm_gMtfBias

// --- 2. Dynamic Regime-Adaptive Weighting ------------------------------------
// Regime only re-weights DIRECTIONAL evidence. It never flips a sign and never
// adds a one-sided bonus.
float curW_trend  = tm_w_trend
float curW_struct = tm_w_struct
float curW_liq    = tm_w_liq
float curW_loc    = tm_w_loc
float curW_sr     = tm_w_sr
float curW_mom    = tm_w_mom

if tm_regime.code == 1 // TRENDING REGIME
    curW_trend  := tm_w_trend  * 1.30
    curW_struct := tm_w_struct * 1.25
    curW_mom    := tm_w_mom    * 1.20
    curW_sr     := tm_w_sr     * 0.60
else if tm_regime.code == 0 // RANGING REGIME
    curW_trend  := tm_w_trend  * 0.60
    curW_sr     := tm_w_sr     * 1.30
    curW_liq    := tm_w_liq    * 1.30
    curW_loc    := tm_w_loc    * 1.20
else if tm_regime.code == -1 // CHOPPY SQUEEZE
    curW_trend  := tm_w_trend  * 0.50
    curW_struct := tm_w_struct * 0.50
    curW_mom    := tm_w_mom    * 0.50

// --- 3. Ablation Feature Config (v2.1 protocol A-E) --------------------------
// A: Structure + Location. B: A + Trend/Momentum. C: B + Volume.
// D: C + MTF/Session. E: full feature set.
// Note: E is about FEATURES. Execution gates remain the explicit toggles in
// "TM · Eligibility Gates" so an ablation cannot silently disarm safety rules.
int tm_cfgCode = tm_ablationMode == "A (Structure + Location)" ? 1 : tm_ablationMode == "B (A + Trend/Momentum)" ? 2 : tm_ablationMode == "C (B + Volume)" ? 3 : tm_ablationMode == "D (C + MTF + Regime)" ? 4 : 5
string tm_cfgName = tm_cfgCode == 1 ? "A" : tm_cfgCode == 2 ? "B" : tm_cfgCode == 3 ? "C" : tm_cfgCode == 4 ? "D" : "E"

float cfgTrend  = tm_cfgCode >= 2 ? curW_trend : 0.0
float cfgStruct = curW_struct
float cfgLiq    = curW_liq
float cfgLoc    = curW_loc
float cfgSr     = curW_sr
float cfgCandle = tm_w_candle
float cfgVol    = tm_cfgCode >= 3 ? tm_w_vol : 0.0
float cfgMom    = tm_cfgCode >= 2 ? curW_mom : 0.0
float cfgSess   = tm_cfgCode >= 4 ? tm_w_sess : 0.0
float cfgMtf    = tm_cfgCode >= 4 ? tm_w_mtf : 0.0

// --- 4. Symmetric Side Scores (0..100) ---------------------------------------
float tm_wDirTotal = cfgTrend + cfgStruct + cfgLiq + cfgLoc + cfgSr + cfgCandle + cfgVol + cfgMom + cfgSess + cfgMtf

float tm_longWeighted = cfgTrend  * f_tmLongPart(tm_gTrend) + cfgStruct * f_tmLongPart(tm_gStruct) + cfgLiq * f_tmLongPart(tm_gLiq) + cfgLoc * f_tmLongPart(tm_gLoc) + cfgSr * f_tmLongPart(tm_gSr) + cfgCandle * f_tmLongPart(tm_gCandle) + cfgVol * f_tmLongPart(tm_gVolume) + cfgMom * f_tmLongPart(tm_gMom) + cfgSess * f_tmLongPart(tm_gSess) + cfgMtf * f_tmLongPart(tm_gMtf)
float tm_shortWeighted = cfgTrend * f_tmShortPart(tm_gTrend) + cfgStruct * f_tmShortPart(tm_gStruct) + cfgLiq * f_tmShortPart(tm_gLiq) + cfgLoc * f_tmShortPart(tm_gLoc) + cfgSr * f_tmShortPart(tm_gSr) + cfgCandle * f_tmShortPart(tm_gCandle) + cfgVol * f_tmShortPart(tm_gVolume) + cfgMom * f_tmShortPart(tm_gMom) + cfgSess * f_tmShortPart(tm_gSess) + cfgMtf * f_tmShortPart(tm_gMtf)

int tm_longScore100  = f_tmSideScore(tm_longWeighted, tm_wDirTotal)
int tm_shortScore100 = f_tmSideScore(tm_shortWeighted, tm_wDirTotal)

// --- 5. Direction Selection (minimum side score + margin) --------------------
bool   tm_longMinOk  = tm_longScore100 > tm_scr_noTrade
bool   tm_shortMinOk = tm_shortScore100 > tm_scr_noTrade
int    tm_scoreGap   = math.abs(tm_longScore100 - tm_shortScore100)
string tm_directionCode = not (tm_longMinOk or tm_shortMinOk) ? "NO_DIRECTION" : tm_longMinOk and tm_shortMinOk and tm_scoreGap < tm_scoreMargin ? "AMBIGUOUS_SETUP" : "OK"
int    tm_candidateDir  = tm_longMinOk and tm_longScore100 >= tm_shortScore100 ? 1 : tm_shortMinOk and tm_shortScore100 > tm_longScore100 ? -1 : 0
int    tm_dir           = tm_directionCode == "OK" ? tm_candidateDir : 0

// Backwards-compatible aliases used by the planner, state machine and visuals.
// `tm_score100` now means "score of the selected side" (never a signed quality).
int tm_score100 = tm_dir == 1 ? tm_longScore100 : tm_dir == -1 ? tm_shortScore100 : math.max(tm_longScore100, tm_shortScore100)

// A directionless/ambiguous bar is not a tradeable quality, even when one side
// happens to be strong. Report NO TRADE so the dashboard cannot show "A+"
// next to NO_DIRECTION/AMBIGUOUS_SETUP.
string tm_quality      = tm_directionCode != "OK" ? "NO TRADE" : f_tmLevelName(f_tmScoreLevel(tm_score100, tm_scr_noTrade, tm_scr_weak, tm_scr_setup, tm_scr_strong))
int    tm_qualityLevel = tm_directionCode != "OK" ? 0 : f_tmScoreLevel(tm_score100, tm_scr_noTrade, tm_scr_weak, tm_scr_setup, tm_scr_strong)
int    tm_minReqLevel  = f_tmMinLevel(tm_minQuality)
bool   tm_okQuality    = tm_qualityLevel >= tm_minReqLevel

// Single score-side gate code. Empty direction and low quality are DIFFERENT
// states: "no setup" is not "blocked".
string tm_scoreBlockCode = tm_directionCode != "OK" ? tm_directionCode : not tm_okQuality ? (tm_dir == 1 ? "BLOCKED_SCORE_LONG" : tm_dir == -1 ? "BLOCKED_SCORE_SHORT" : "NO_DIRECTION") : "OK"

string tm_scoreMeta = "v" + tm_scoreVersion + "|cfg" + tm_cfgName + "|L" + str.tostring(tm_longScore100) + "|S" + str.tostring(tm_shortScore100) + "|m" + str.tostring(tm_scoreGap)

// =============================================================================
//  L7b - ENTRY MODES BY MARKET STATE (v2.1 Phase 3)
// =============================================================================
//  Each mode publishes: activation, confirmation, invalidation and expiry. The
//  state machine only accepts a NEW trade when an enabled mode is ready AND its
//  direction matches the direction-aware score. Modes never override risk gates.
//
//  A - Trend Pullback: trending regime, HTF not opposed, pullback into EMA/FVG/
//      OB, then a close back in the trend direction.
//  B - Liquidity Sweep Reversal: sweep of a tracked level, then a rejection /
//      engulfing confirmation within a bounded window after the sweep.
//  C - Range Mean-Reversion: RANGING regime, price at a range edge, rejection
//      candle back inside the range. Disabled by default until validated OOS.
// =============================================================================

float tm_atrSafe = nz(tm_atr, close * 0.01)
float tm_pbTol   = tm_atrSafe * tm_modePullTol
float tm_rgTol   = tm_atrSafe * tm_modeRangeTol

// --- Mode A: Trend Pullback --------------------------------------------------
int  tm_modeADir   = tm_regime.code == 1 ? tm_regime.trendDir : 0
bool tm_modeAMtfOk = tm_modeADir == 1 ? tm_ctxDir >= 0 and tm_setupDir >= 0 : tm_modeADir == -1 ? tm_ctxDir <= 0 and tm_setupDir <= 0 : false

bool tm_modeAPullLong  = (low <= tm_emaF + tm_pbTol) or (not na(tm_nearFvgBullPx) and low <= tm_nearFvgBullPx + tm_pbTol) or (not na(tm_nearObBullPx) and low <= tm_nearObBullPx + tm_pbTol)
bool tm_modeAPullShort = (high >= tm_emaF - tm_pbTol) or (not na(tm_nearFvgBearPx) and high >= tm_nearFvgBearPx - tm_pbTol) or (not na(tm_nearObBearPx) and high >= tm_nearObBearPx - tm_pbTol)
bool tm_modeAConfirmLong  = close > open and close >= tm_emaF - tm_pbTol
bool tm_modeAConfirmShort = close < open and close <= tm_emaF + tm_pbTol

bool tm_modeAReadyLong  = tm_modeA_on and tm_modeADir == 1 and tm_modeAMtfOk and tm_modeAPullLong and tm_modeAConfirmLong
bool tm_modeAReadyShort = tm_modeA_on and tm_modeADir == -1 and tm_modeAMtfOk and tm_modeAPullShort and tm_modeAConfirmShort

// --- Mode B: Liquidity Sweep Reversal ---------------------------------------
int tm_sweepAgeBull = tm_lastSweepBullBar >= 0 ? bar_index - tm_lastSweepBullBar : 999999
int tm_sweepAgeBear = tm_lastSweepBearBar >= 0 ? bar_index - tm_lastSweepBearBar : 999999
bool tm_sweepFreshBull = tm_sweepAgeBull <= tm_modeSweepBars and not na(tm_lastSweepBullPx)
bool tm_sweepFreshBear = tm_sweepAgeBear <= tm_modeSweepBars and not na(tm_lastSweepBearPx)
bool tm_modeBConfirmLong  = tm_bullPin or tm_bullEng or tm_bullRej
bool tm_modeBConfirmShort = tm_bearPin or tm_bearEng or tm_bearRej

bool tm_modeBReadyLong  = tm_modeB_on and tm_sweepFreshBull and tm_modeBConfirmLong and close > tm_lastSweepBullPx
bool tm_modeBReadyShort = tm_modeB_on and tm_sweepFreshBear and tm_modeBConfirmShort and close < tm_lastSweepBearPx

// Invalidation: price accepts beyond the swept extreme instead of rejecting it.
bool tm_modeBInvalidLong  = tm_modeB_on and tm_sweepFreshBull and close < tm_lastSweepBullPx - 0.1 * tm_atrSafe
bool tm_modeBInvalidShort = tm_modeB_on and tm_sweepFreshBear and close > tm_lastSweepBearPx + 0.1 * tm_atrSafe

// --- Mode C: Range Mean-Reversion -------------------------------------------
bool tm_rangeBoxOk = tm_regime.code == 0 and not na(tm_nearestRes) and not na(tm_nearestSup) and (tm_nearestRes - tm_nearestSup) > tm_atrSafe
bool tm_modeCNearSup = tm_rangeBoxOk and close <= tm_nearestSup + tm_rgTol and close >= tm_nearestSup - tm_atrSafe
bool tm_modeCNearRes = tm_rangeBoxOk and close >= tm_nearestRes - tm_rgTol and close <= tm_nearestRes + tm_atrSafe
bool tm_modeCReadyLong  = tm_modeC_on and tm_modeCNearSup and (tm_bullPin or tm_bullEng or tm_bullRej) and close > open
bool tm_modeCReadyShort = tm_modeC_on and tm_modeCNearRes and (tm_bearPin or tm_bearEng or tm_bearRej) and close < open

// --- Mode Selection (A > B > C) ---------------------------------------------
// Priority is deterministic and documented so the same bar always resolves to
// the same mode. C stays opt-in until the range classifier is validated OOS.
int    tm_modeDir    = 0
string tm_activeMode = "NONE"

if tm_modeAReadyLong
    tm_modeDir    := 1
    tm_activeMode := "A_TREND_PULLBACK"
else if tm_modeAReadyShort
    tm_modeDir    := -1
    tm_activeMode := "A_TREND_PULLBACK"
else if tm_modeBReadyLong
    tm_modeDir    := 1
    tm_activeMode := "B_LIQUIDITY_SWEEP"
else if tm_modeBReadyShort
    tm_modeDir    := -1
    tm_activeMode := "B_LIQUIDITY_SWEEP"
else if tm_modeCReadyLong
    tm_modeDir    := 1
    tm_activeMode := "C_RANGE_REVERSION"
else if tm_modeCReadyShort
    tm_modeDir    := -1
    tm_activeMode := "C_RANGE_REVERSION"

bool tm_modeReady = tm_modeDir != 0

// Why no mode is active (stable priority, used by the dashboard/alerts).
bool tm_modeAwaitConfirm = tm_modeReady or tm_modeAPullLong or tm_modeAPullShort or tm_sweepFreshBull or tm_sweepFreshBear or tm_modeCNearSup or tm_modeCNearRes
string tm_modeReason = not tm_modeFilterOn ? "OK" : tm_modeReady ? (tm_activeMode == "A_TREND_PULLBACK" ? "MODE_A_PULLBACK" : tm_activeMode == "B_LIQUIDITY_SWEEP" ? "MODE_B_SWEEP" : "MODE_C_RANGE") : tm_modeBInvalidLong or tm_modeBInvalidShort ? "MODE_INVALIDATED" : tm_modeAwaitConfirm ? "WAIT_CONFIRMATION" : tm_modeADir != 0 ? "WAIT_PULLBACK" : "NO_SETUP"

// The mode is a gate, not a score bonus: a great score cannot bypass it.
bool tm_modeGateOk = not tm_modeFilterOn or (tm_modeReady and tm_modeDir == tm_dir)

// =============================================================================
//  L8 - PLAN BUILDER & DYNAMIC STATE MACHINE (v2.1)
// =============================================================================
//  v2.1 changes:
//    * RR gate names the target it evaluates (TP2) in the plan, dashboard and
//      alert payload, and emits BLOCKED_RR / BLOCKED_NET_RR codes.
//    * Every rejection is a standardized reason code; the full blocker set is
//      kept separately so the Advanced dashboard/alert can show all causes.
//    * Entry modes (A/B/C) are an independent gate: confidence cannot bypass
//      an inactive mode, and a mode cannot bypass risk.
//    * Lifecycle counters distinguish candidate -> eligible -> pending ->
//      filled -> closed, so a blocked setup is never reported as a trade.
// =============================================================================

// --- 1. Comprehensive Trade Plan Builder -------------------------------------
f_buildTMPlan2(int dir) =>
    float a     = tm_atr
    bool  atrOk = not na(a) and a > 0.0
    bool  dirOk = dir == 1 or dir == -1

    // Determine Entry Type (Market vs Pullback Limit)
    string ordType = "MARKET"
    if tm_entryMode == "Pullback Limit (Strict)"
        ordType := "LIMIT"
    else if tm_entryMode == "Auto (Adaptive)"
        if tm_isOverextended
            ordType := "LIMIT"
        else
            ordType := "MARKET"

    // Limit Price Selection
    float limPx = na
    if dir == 1
        limPx := tm_limitSource == "50% FVG" and not na(tm_nearFvgBullPx) ? tm_nearFvgBullPx : tm_limitSource == "Order Block Edge" and not na(tm_nearObBullPx) ? tm_nearObBullPx : tm_emaF
        // Safety: a buy limit must sit at or below the current close
        limPx := math.min(limPx, close - syminfo.mintick)
    else if dir == -1
        limPx := tm_limitSource == "50% FVG" and not na(tm_nearFvgBearPx) ? tm_nearFvgBearPx : tm_limitSource == "Order Block Edge" and not na(tm_nearObBearPx) ? tm_nearObBearPx : tm_emaF
        // Safety: a sell limit must sit at or above the current close
        limPx := math.max(limPx, close + syminfo.mintick)

    float entryPx = ordType == "LIMIT" ? limPx : close

    // SL Distance & Anchoring
    float pad   = a * tm_sl_padAtr
    float swing = dir == 1 ? nz(tm_swingLo, low) : nz(tm_swingHi, high)
    float sDist = dir == 1 ? (entryPx - swing) + pad : (swing - entryPx) + pad
    float aDist = a * tm_sl_atrMul
    float base  = (aDist + sDist) * 0.5
    float dist  = f_clamp(nz(base, aDist), a * tm_sl_minAtr, a * tm_sl_maxAtr)

    float slPx = entryPx - dir * dist
    float risk = dist

    // Targets: TP1 (partial + auto-BE), TP2 (standard, evaluated vs minRR), TP3 (runner)
    float r1  = tm_rr1
    float r2  = math.max(tm_rr2, r1 + 0.5)
    float r3  = math.max(tm_rr3, r2 + 1.0)
    float tp1 = entryPx + dir * r1 * risk
    float tp2 = entryPx + dir * r2 * risk
    float tp3 = entryPx + dir * r3 * risk

    // RR gate is evaluated at TP2, never at TP1 (partial target).
    float evalRR = r2
    bool  rrOk   = tm_rrMin <= 0.0 or evalRR >= tm_rrMin
    bool  slOk   = dist > 0.0 and not na(slPx) and (dir == 1 ? slPx < entryPx : slPx > entryPx)
    bool  tpOk   = not na(tp1) and not na(tp2) and not na(tp3) and (dir == 1 ? (entryPx < tp1 and tp1 < tp2 and tp2 < tp3) : (entryPx > tp1 and tp1 > tp2 and tp2 > tp3))

    // Futures Position Sizing & Margin
    float riskUsd   = tm_risk_equity * (tm_risk_pct / 100.0)
    float posSize   = dist > 0.0 ? riskUsd / dist : 0.0
    float notional  = posSize * entryPx
    float effLev    = tm_risk_equity > 0.0 ? notional / tm_risk_equity : 0.0
    float marginReq = tm_risk_maxLev > 0.0 ? notional / tm_risk_maxLev : notional

    // Estimated Liquidation Price
    float levForLiq = math.max(effLev, 1.0)
    float liqLong   = entryPx * math.max(0.0, 1.0 - (1.0 / levForLiq) + tm_risk_mmr)
    float liqShort  = entryPx * (1.0 + (1.0 / levForLiq) - tm_risk_mmr)
    float liqPrice  = dir == 1 ? liqLong : dir == -1 ? liqShort : na

    bool  liqOk     = not tm_gate_risk or (dir == 1 ? slPx > liqPrice : slPx < liqPrice)
    bool  levOk     = not tm_gate_risk or tm_risk_maxLev <= 0.0 or effLev <= (tm_risk_maxLev * 1.05)

    // Friction & Net RR
    float feeSlipUsd = notional * (tm_risk_feeSlipPct / 100.0)
    float netProfit  = (evalRR * riskUsd) - feeSlipUsd
    float netRisk    = riskUsd + feeSlipUsd
    float netRR      = netRisk > 0.0 ? netProfit / netRisk : evalRR
    // Net RR is compared against the SAME minimum target, after fees/slippage.
    // (v2.0 used minRR * 0.70, which hid the cost of the trade.)
    bool  netRrOk    = tm_rrMin <= 0.0 or netRR >= tm_rrMin

    // Standardized plan reason codes (v2.1 section 11.3).
    string rsn = not atrOk ? "BLOCKED_NO_ATR" : not dirOk ? "NO_DIRECTION" : not slOk ? "BLOCKED_BAD_SL" : not tpOk ? "BLOCKED_BAD_TP" : not rrOk ? "BLOCKED_RR" : not levOk ? "BLOCKED_MAX_LEV" : not liqOk ? "BLOCKED_LIQUIDATION_RISK" : not netRrOk ? "BLOCKED_NET_RR" : "VALID"
    bool  valid = atrOk and dirOk and slOk and tpOk and rrOk and levOk and liqOk and netRrOk

    TMPlan2.new(valid, dir, ordType, entryPx, limPx, slPx, tp1, tp2, tp3, risk, evalRR, posSize, notional, effLev, marginReq, liqPrice, netRR, "TP2", rsn)

tm_plan2 = f_buildTMPlan2(tm_dir)

tm_barsSince += 1

var int   tm_statTimeouts       = 0

var float tm_statRealizedRSum   = 0.0

// Lifecycle funnel counters (v2.1 section 13.1)

var int tm_statEligible   = 0
var int tm_statPendingN   = 0

var bool tm_prevCandidate = false
var bool tm_prevEligible  = false

// --- 3. Gate Status & Eligibility --------------------------------------------
tm_gateStatus = f_evaluateGates(tm_dir)

bool tm_okBar     = tm_confirmOnly ? barstate.isconfirmed : true
bool tm_okCool    = tm_barsSince >= tm_coolBars
bool tm_okDay     = tm_dayCount < tm_maxDay
bool tm_okDir     = tm_dir != 0
bool tm_okCircuit = not tm_gate_circuit or tm_consecLosses < tm_risk_maxLosses
bool tm_okPosition = tm_posStage == TM2_STAGE_IDLE

bool tm_tradeEligible = tm_okDir and tm_okQuality and tm_gateStatus.allPassed and tm_modeGateOk and tm_okCool and tm_okDay and tm_okCircuit and tm_okPosition
bool tm_riskEligible  = tm_plan2.valid
bool tm_signal        = tm_okBar and tm_tradeEligible and tm_riskEligible

if tm_scoreBlockCode != "OK"

if not tm_gateStatus.allPassed
    tm_blockMask := f_tmBlockers(tm_blockMask, tm_gateStatus.blockMask)
if not tm_modeGateOk
    tm_blockMask := f_tmBlockers(tm_blockMask, tm_modeReason)
if not tm_okCool
    tm_blockMask := f_tmBlockers(tm_blockMask, "BLOCKED_COOLDOWN")
if not tm_okDay
    tm_blockMask := f_tmBlockers(tm_blockMask, "BLOCKED_DAILY_LIMIT")
if not tm_okCircuit
    tm_blockMask := f_tmBlockers(tm_blockMask, "BLOCKED_CIRCUIT_BREAKER")
if not tm_okPosition
    tm_blockMask := f_tmBlockers(tm_blockMask, "BLOCKED_POSITION_OPEN")
if tm_dir != 0 and not tm_riskEligible
    tm_blockMask := f_tmBlockers(tm_blockMask, tm_plan2.reason)

string tm_noTradeReason = tm_scoreBlockCode != "OK" ? tm_scoreBlockCode : not tm_gateStatus.regimeOk ? "BLOCKED_REGIME" : not tm_gateStatus.mtfOk ? "BLOCKED_MTF" : not tm_gateStatus.btcOk ? (tm_dir == 1 ? "BTC_DUMPING_VETO" : "BTC_PUMPING_VETO") : not tm_gateStatus.obstacleOk ? "BLOCKED_OBSTACLE" : not tm_gateStatus.overextOk ? "BLOCKED_OVEREXTENSION" : not tm_modeGateOk ? tm_modeReason : not tm_okCool ? "BLOCKED_COOLDOWN" : not tm_okDay ? "BLOCKED_DAILY_LIMIT" : not tm_okCircuit ? "BLOCKED_CIRCUIT_BREAKER" : not tm_okPosition ? "BLOCKED_POSITION_OPEN" : not tm_riskEligible ? tm_plan2.reason : "OK"

// Eligible = candidate that survived every gate (still not necessarily filled).

tm_evPending := false

// --- 5. Dynamic State Machine ------------------------------------------------
// 1. Deferred reset from the previous bar (keeps the exit label readable).
if tm_needReset
    tm_needReset  := false
    tm_posStage   := TM2_STAGE_IDLE
    tm_posDir     := 0
    tm_posType    := "MARKET"
    tm_posMode    := ""
    tm_posEntry   := na
    tm_posLimit   := na
    tm_posSl      := na
    tm_posTp1     := na
    tm_posTp2     := na
    tm_posTp3     := na
    tm_posRisk    := na
    tm_posBars    := 0
    tm_posBar     := na
    tm_posMaxFavR := 0.0
    tm_exitReason := TM2_EXIT_NONE

// 2. Manage PENDING_LIMIT orders
if tm_posStage == TM2_STAGE_PENDING_LIMIT
    tm_posBars += 1
    bool isUp = tm_posDir == 1
    bool filled = isUp ? low <= tm_posLimit : high >= tm_posLimit
    bool reachedTp1Early = isUp ? high >= tm_posTp1 : low <= tm_posTp1

    if filled
        tm_posStage := TM2_STAGE_OPEN
        tm_posEntry := tm_posLimit
        tm_posBar   := bar_index
        tm_posBars  := 0
        tm_evFilled := true
        tm_statFilledN += 1
    else if tm_limitCancelOnTp1 and reachedTp1Early
        // Missed move: price reached TP1 before filling the limit.
        tm_exitReason := TM2_CANCEL_MISSED
        tm_evCancel   := true
        tm_needReset  := true
    else if tm_posBars >= tm_limitCancelBars
        // Expired limit order after K bars.
        tm_exitReason := TM2_CANCEL_TIMEOUT
        tm_evCancel   := true
        tm_needReset  := true

// 3. Manage Active OPEN Position (TP1 BE, TP2 trail, time-decay, timeout)
if tm_posStage >= TM2_STAGE_OPEN
    tm_posBars += 1
    bool  isUp = tm_posDir == 1
    float curFavR = isUp ? (high - tm_posEntry) / tm_posRisk : (tm_posEntry - low) / tm_posRisk
    tm_posMaxFavR := math.max(tm_posMaxFavR, curFavR)

    float t1 = tm_posStage < TM2_STAGE_TP1_BE ? tm_posTp1 : na
    float t2 = tm_posStage < TM2_STAGE_TP2_TRAIL ? tm_posTp2 : na
    float t3 = tm_posTp3

    // A limit that fills on this bar may have filled AFTER the high/low that
    // touched a target. Without intrabar data, counting a TP on the fill bar is
    // optimistic, so targets are only evaluated from the next bar onward. A stop
    // on the fill bar is still honored (worst-case assumption).
    bool tm_fillBar = tm_posBar == bar_index
    bool hitSl = isUp ? low <= tm_posSl : high >= tm_posSl
    bool hit1  = not tm_fillBar and not na(t1) and (isUp ? high >= t1 : low <= t1)
    bool hit2  = not tm_fillBar and not na(t2) and (isUp ? high >= t2 : low <= t2)
    bool hit3  = not tm_fillBar and not na(t3) and (isUp ? high >= t3 : low <= t3)
    bool hitAnyTp = hit1 or hit2 or hit3

    // TP1 hit rate counts a trade once, only when TP1 was genuinely reachable
    // and the conservative SL-first rule did not fire on the same bar.
    if hitAnyTp and tm_posStage == TM2_STAGE_OPEN and not hitSl
        tm_statTp1Hits += 1

    bool closed = false
    float realizedTradeR = 0.0

    // Conservative execution: if a bar hits both SL and TP, SL takes precedence.
    if hitSl and hitAnyTp
        closed := true
        tm_evSl := true
        tm_evClose := true
        if tm_posStage == TM2_STAGE_OPEN
            tm_exitReason := TM2_EXIT_SL
            realizedTradeR := -1.0
            tm_statFullLosses += 1
            tm_consecLosses   += 1
        else if tm_posStage == TM2_STAGE_TP1_BE
            tm_exitReason := TM2_EXIT_BREAKEVEN
            // 40% at TP1 + 60% at BE (0R) minus roundtrip fee buffer
            realizedTradeR := 0.40 * tm_rr1 - (tm_risk_feeSlipPct * 2.0 / 100.0)
            tm_statBeSaves += 1
            tm_statNetWins += 1
            tm_consecLosses := 0
        else
            tm_exitReason := TM2_EXIT_TRAIL
            realizedTradeR := 0.40 * tm_rr1 + 0.40 * tm_rr2 - (tm_risk_feeSlipPct * 2.0 / 100.0)
            tm_statNetWins += 1
            tm_consecLosses := 0
    else if hitSl
        closed := true
        tm_evSl := true
        tm_evClose := true
        if tm_posStage == TM2_STAGE_OPEN
            tm_exitReason := TM2_EXIT_SL
            realizedTradeR := -1.0
            tm_statFullLosses += 1
            tm_consecLosses   += 1
        else if tm_posStage == TM2_STAGE_TP1_BE
            tm_exitReason := TM2_EXIT_BREAKEVEN
            realizedTradeR := 0.40 * tm_rr1 - (tm_risk_feeSlipPct * 2.0 / 100.0)
            tm_statBeSaves += 1
            tm_statNetWins += 1
            tm_consecLosses := 0
        else
            tm_exitReason := TM2_EXIT_TRAIL
            realizedTradeR := 0.40 * tm_rr1 + 0.40 * tm_rr2 - (tm_risk_feeSlipPct * 2.0 / 100.0)
            tm_statNetWins += 1
            tm_consecLosses := 0
    else if hit3
        closed := true
        tm_evTp := 3
        tm_evClose := true
        tm_exitReason := TM2_EXIT_TP3
        realizedTradeR := 0.40 * tm_rr1 + 0.40 * tm_rr2 + 0.20 * tm_rr3
        tm_statNetWins += 1
        tm_consecLosses := 0
    else if hit2
        tm_evTp := 2
        tm_posStage := TM2_STAGE_TP2_TRAIL
        // Lock TP1 as the structural base for the trail
        tm_posSl := tm_posTp1
        tm_evTrail := true
    else if hit1
        tm_evTp := 1
        tm_posStage := TM2_STAGE_TP1_BE
        // Auto Break-Even: move SL to entry +/- fee buffer
        if tm_autoBeOn
            tm_posSl := tm_posEntry + tm_posDir * tm_atr * tm_beOffsetAtr
            tm_evBeMoved := true

    // Dynamic trailing stop in TP2 stage
    if not closed and tm_posStage == TM2_STAGE_TP2_TRAIL and tm_trailOn
        float trailLvl = isUp ? high[1] - tm_trailDistAtr * tm_atr : low[1] + tm_trailDistAtr * tm_atr
        tm_posSl := isUp ? math.max(tm_posSl, trailLvl) : math.min(tm_posSl, trailLvl)

    // Time-Decay / Stagnation exit
    if not closed and tm_timeDecayOn and tm_posStage == TM2_STAGE_OPEN and tm_posBars >= tm_stagnantBars
        if tm_posMaxFavR < 0.3 and tm_regime.adx < tm_regimeAdxTrending
            closed := true
            tm_evDecay := true
            tm_evClose := true
            tm_exitReason := TM2_EXIT_TIMEDECAY
            tm_statTimeDecays += 1
            realizedTradeR := (close - tm_posEntry) * tm_posDir / tm_posRisk
            if realizedTradeR > 0
                tm_statNetWins += 1
                tm_consecLosses := 0
            else
                tm_consecLosses += 1

    // Max-hold timeout
    if not closed and tm_posBars >= tm_maxHoldBars
        closed := true
        tm_evTimeout := true
        tm_evClose := true
        tm_exitReason := TM2_EXIT_TIMEOUT
        tm_statTimeouts += 1
        realizedTradeR := (close - tm_posEntry) * tm_posDir / tm_posRisk
        if realizedTradeR > 0
            tm_statNetWins += 1
            tm_consecLosses := 0
        else
            tm_consecLosses += 1

    if closed
        tm_statTotalTrades += 1
        tm_statClosedN += 1
        tm_statRealizedRSum += realizedTradeR
        tm_needReset := true

// 4. Open a new position (one position at a time)
if tm_okBar and tm_signal and tm_posStage == TM2_STAGE_IDLE
    tm_posDir    := tm_plan2.dir
    tm_posType   := tm_plan2.orderType
    tm_posMode   := tm_activeMode
    tm_posEntry  := tm_plan2.entry
    tm_posLimit  := tm_plan2.limitPrice
    tm_posSl     := tm_plan2.sl
    tm_posTp1    := tm_plan2.tp1
    tm_posTp2    := tm_plan2.tp2
    tm_posTp3    := tm_plan2.tp3
    tm_posRisk   := tm_plan2.risk
    tm_posBars   := 0
    tm_posBar    := bar_index
    tm_posMaxFavR:= 0.0
    tm_posSigId  := "TM2_" + syminfo.ticker + "_" + timeframe.period + "_" + str.tostring(time) + "_" + (tm_plan2.dir == 1 ? "L" : "S")
    tm_dayCount  += 1
    tm_barsSince := 0
    tm_exitReason:= TM2_EXIT_NONE

    if tm_plan2.orderType == "LIMIT"
        tm_posStage := TM2_STAGE_PENDING_LIMIT
        tm_evPending := true
        tm_statPendingN += 1
    else
        tm_posStage := TM2_STAGE_OPEN
        tm_evEntry  := true
        tm_statFilledN += 1

    table.cell(t, 0, row, key, text_color = keyCol, text_size = fSize, text_halign = text.align_left, bgcolor = bgCol)
    table.cell(t, 1, row, val, text_color = valCol, text_size = fSize, text_halign = text.align_right, bgcolor = bgCol)
